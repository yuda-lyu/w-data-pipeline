import { normalizeLogger } from '../util/logger.mjs'
import { resolveIdentity } from './identity.mjs'
import { openCollection } from './openCollection.mjs'
import { ContractError } from '../core/errors.mjs'

/**
 * 建立去重檢測器：以主鍵判斷是否已抓過。
 *
 * 【兩種判定，用途不同，兩個都要】peek() 只判斷不寫入——用於「抓之前先問，
 * 貴的動作（補抓內文數十秒、送 AI 數分鐘且燒額度）不要白做」；
 * admit() 判斷並占位（實際 insert）——用於「放行閘門」，這是真正的去重。
 * 只有 peek 不行：它是「先查再寫」的前半，中間有空窗，同一輪的併發批次會在
 * 空窗內雙雙判為未抓過而重複放行；admit 沒有這個空窗——判定與占位由儲存層
 * 在同一寫交易內原子完成（如 w-orm-lmdb 之 ifNoExists，實測併發 100 次插入
 * 同一 id 僅一次成功）。故要判定「是不是新資料」一律以 admit 為準，peek 只作預篩。
 * 只有 admit 也不行：決定「這一篇要不要花錢抓內文」時還不該入庫
 * （可能根本抓不到而該留給下輪重試），此時要用 peek。
 *
 * 【檢測器就是資料庫本身，不另建索引】記錄存在即去重憑證（終態文件可瘦身清掉
 * 大欄位但記錄不刪，去重即持續成立）；若另建索引，就會多出一份要同步的真相，
 * 而兩份真相遲早不一致。
 *
 * 【身分定義由 identity 物件承擔】本層不假設「用 url 判斷」——要用網址、
 * API 流水號、還是自訂字串，全由傳入的 identity 決定（見 dedup/identity.mjs）。
 * 契約層與本層共用同一個 identity 物件，結構上杜絕兩邊算出不同鍵。
 *
 * @param {object} opt
 * @param {object} [opt.collection]  已開啟的集合（openCollection 之產出）；與 lmdb 二擇一
 * @param {{WOrm:Function, path:string, db:string, cl:string}} [opt.lmdb] 由本函數自行開啟集合；
 *     WOrm 為注入之 ORM 建構函數（見 openCollection 之「WOrm 由外部注入」說明）
 * @param {object} [opt.identity]    createIdentity 之產出；未給則以下列欄位就地建立
 * @param {string|Function} [opt.keyOf]      主鍵策略（identity 未給時必填）
 * @param {Function} [opt.identityOf]  ((item)=>string)  自然鍵取用函數（預設 key → canonicalUrl → url）
 * @param {object} [opt.keyOpt]
 * @param {object} [opt.log]
 * @returns {object} 檢測器：{ collection, identity, idOf, has, peek, admit, mark, count, close }
 */
export function createSeenStore(opt = {}) {
    const log = normalizeLogger(opt.log)
    const identity = resolveIdentity(opt.identity || { keyOf: opt.keyOf, identityOf: opt.identityOf, keyOpt: opt.keyOpt })

    let owned = false
    let col = opt.collection
    if (!col) {
        if (!opt.lmdb) throw new ContractError('createSeenStore 需要 collection 或 lmdb 其中之一')
        col = openCollection(opt.lmdb)
        owned = true // 自己開的自己關；外部傳進來的由外部管生命週期
    }

    /**
   * 把項目分成「可識別」與「無效」兩堆。
   * 【為何無自然鍵者一律不放行】其主鍵會是空字串，所有無自然鍵者共用同一個 LMDB key，
   *   造成靜默的錯誤去重（第二筆之後全被判為重複）。
   */
    const split = (items) => {
        const usable = []
        const invalid = []
        for (const it of items || []) {
            const id = identity.of(it)
            if (!id) invalid.push({ item: it, reason: '無法取得自然鍵（key／canonicalUrl／url 皆空）' })
            else usable.push({ item: it, id })
        }
        return { usable, invalid }
    }

    return {
        collection: col,
        /** 本檢測器採用的身分定義（供契約層共用，確保兩層算出同一組鍵） */
        identity,
        /** 由項目或自然鍵字串算出主鍵 */
        idOf: (x) => identity.of(x),

        /** 單一項目／自然鍵是否已抓過（純查詢，不寫入） */
        async has(x) {
            const id = identity.of(x)
            if (!id) return false
            return !!(await col.get(id))
        },

        /**
     * 批次判斷，不寫入。
     * @returns {Promise<{fresh:Array, dup:Array, invalid:Array}>} fresh／dup 內為原始項目
     */
        async peek(items) {
            const { usable, invalid } = split(items)
            const fresh = []
            const dup = []
            for (const u of usable) {
                if (await col.get(u.id)) dup.push(u.item)
                else fresh.push(u.item)
            }
            return { fresh, dup, invalid }
        },

        /**
     * 放行閘門：判斷並占位。只有從未見過的項目會被寫入並回傳於 fresh。
     *
     * @param {Array} items
     * @param {object} [o]
     * @param {Function} [o.toRecord]  ((item, meta)=>object) 入庫前補業務欄位（status／collectedAt／
     *        fetchTries…）；meta 帶 { id, source, fetcher } 等脈絡。未給則原樣入庫。
     *        無論如何，id 一律由本函數覆寫為主鍵——這是去重能成立的前提。
     * @param {object} [o.meta] 額外脈絡，原樣併入傳給 toRecord 的 meta
     * @returns {Promise<{fresh:Array, dup:number, invalid:Array}>}
     *          fresh 為「實際寫入的記錄」（已含 id），可直接拿去做後續處理
     */
        async admit(items, o = {}) {
            const { usable, invalid } = split(items)
            const recs = usable.map((u) => {
                const meta = { ...(o.meta || {}), id: u.id }
                const base = typeof o.toRecord === 'function' ? o.toRecord(u.item, meta) : u.item
                return { ...base, id: u.id }
            })
            const r = await col.insertNew(recs)
            if (invalid.length) log.warn(`去重檢測：${invalid.length} 項無法識別而未放行`)
            return { fresh: r.added, dup: r.dupCount, invalid }
        },

        /**
     * 把某個自然鍵直接記為已見（無完整項目時用，例如只知道網址的黑名單）。
     * @param {string|object} x 自然鍵字串或項目
     */
        async mark(x, fields = {}) {
            const id = identity.of(x)
            if (!id) throw new ContractError('mark 無法取得自然鍵')
            const base = typeof x === 'string' ? { key: x } : x
            const r = await col.insertNew([{ ...base, ...fields, id }])
            return r.addCount === 1
        },

        async count() {
            return col.count()
        },

        /** 自行開啟的集合才由本物件關閉；外部注入者不代管（避免關掉呼叫端還在用的連線） */
        async close() {
            if (owned) await col.close()
        },
    }
}

export default createSeenStore
